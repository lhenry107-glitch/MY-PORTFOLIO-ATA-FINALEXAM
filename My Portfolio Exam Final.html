CC102 Final Performance Task (PeTa): Word Scramble Challenge - A console-based word puzzle game featuring four difficulty levels: Easy, Medium, Hard, and Extreme. 

import java.util.Scanner;
import java.util.Random;

public class WordPuzzle {

    static Scanner input = new Scanner(System.in);
    static Random random = new Random();

    public static void main(String[] args) {

        boolean running = true;

        String[] easy = {
            "APPLE", "SCHOOL", "HAPPY", "ORANGE", "PENCIL"
        };

        String[] medium = {
            "COMPUTER", "INTERNET", "KEYBOARD",
            "MONITOR", "LIBRARY"
        };

        String[] hard = {
            "PROGRAMMING", "ALGORITHM", "DATABASE",
            "VARIABLE", "FUNCTION"
        };

        String[] extreme = {
            "TECHNOLOGY", "CYBERSECURITY", "ARCHITECTURE",
            "COMMUNICATION", "SYNCHRONIZATION"
        };

        while (running) {

            System.out.println("\n================================");
            System.out.println("      WORD SCRAMBLE CHALLENGE");
            System.out.println("================================");
            System.out.println("[1] Easy");
            System.out.println("[2] Medium");
            System.out.println("[3] Hard");
            System.out.println("[4] Extreme");
            System.out.println("[5] Exit");
            System.out.println("================================");
            System.out.print("Choose difficulty: ");

            int choice = getNumber();

            switch (choice) {

                case 1:
                    playGame("EASY", easy);
                    break;

                case 2:
                    playGame("MEDIUM", medium);
                    break;

                case 3:
                    playGame("HARD", hard);
                    break;

                case 4:
                    playGame("EXTREME", extreme);
                    break;

                case 5:
                    System.out.println("\nThanks for playing!");
                    System.out.println("Keep learning and keep solving!");
                    running = false;
                    break;

                default:
                    System.out.println("Invalid choice! Please try again.");
            }
        }

        input.close();
    }

    static void playGame(String difficulty, String[] words) {

        int score = 0;
        int correct = 0;
        int wrong = 0;

        System.out.println("\n================================");
        System.out.println("          " + difficulty);
        System.out.println("================================");
        System.out.println("Unscramble the words!");
        System.out.println("You have 5 questions.");
        System.out.println("================================");

        for (int i = 0; i < words.length; i++) {

            String originalWord = words[i];
            String scrambledWord = scramble(originalWord);

            System.out.println("\nQuestion " + (i + 1) + " of 5");
            System.out.println("Unscramble this word: " + scrambledWord);

            System.out.print("Your answer: ");
            String answer = input.nextLine().trim();

            if (answer.equalsIgnoreCase(originalWord)) {

                System.out.println("CORRECT! Great job!");
                score++;
                correct++;

            } else {

                System.out.println("INCORRECT!");
                System.out.println("Correct answer: " + originalWord);
                wrong++;
            }
        }

        System.out.println("\n================================");
        System.out.println("          GAME RESULTS");
        System.out.println("================================");

        System.out.println("Difficulty: " + difficulty);
        System.out.println("Total Questions: 5");
        System.out.println("Correct Answers: " + correct);
        System.out.println("Wrong Answers: " + wrong);
        System.out.println("Your Score: " + score + "/5");
        System.out.println("Percentage: " + (score * 20) + "%");

        if (score == 5) {
            System.out.println("Excellent! Perfect score!");
        } else if (score >= 3) {
            System.out.println("Good job! Keep practicing!");
        } else {
            System.out.println("Don't give up! Try again!");
        }

        System.out.println("================================");
        System.out.println("Returning to main menu...");
    }

    static String scramble(String word) {

        char[] letters = word.toCharArray();

        for (int attempt = 0; attempt < 20; attempt++) {

            for (int i = letters.length - 1; i > 0; i--) {

                int j = random.nextInt(i + 1);

                char temp = letters[i];
                letters[i] = letters[j];
                letters[j] = temp;
            }

            String scrambled = new String(letters);

            if (!scrambled.equals(word)) {
                return scrambled;
            }
        }

        char first = letters[0];

        for (int i = 0; i < letters.length - 1; i++) {
            letters[i] = letters[i + 1];
        }

        letters[letters.length - 1] = first;

        return new String(letters);
    }

    static int getNumber() {

        while (true) {

            String value = input.nextLine();

            try {
                return Integer.parseInt(value);
            } catch (NumberFormatException e) {
                System.out.print("Please enter a valid number: ");
            }
        }
    }
}
